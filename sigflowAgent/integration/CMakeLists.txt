    message(FATAL_ERROR "SIGFLOW_BUILD_U2_INTEGRATION requires SIGFLOW_BUILD_EDU_AGENT=ON")

find_package(Threads REQUIRED)

target_compile_definitions(edu_u2_gateway PRIVATE SIGFLOW_REPO_ROOT="${CMAKE_SOURCE_DIR}")
add_executable(edu_u2_sim_artifact_test EXCLUDE_FROM_ALL sim_artifact_test.cpp)
target_link_libraries(edu_u2_sim_artifact_test PRIVATE eda-sim-verilator eda_core Threads::Threads)
target_include_directories(edu_u2_sim_artifact_test PRIVATE "${CMAKE_SOURCE_DIR}/core/src")
add_executable(edu_u2_wave_service_test EXCLUDE_FROM_ALL wave_service_test.cpp)
target_link_libraries(edu_u2_wave_service_test PRIVATE eda_agent_gateway Threads::Threads)
add_executable(edu_u2_vcd_integrity_test EXCLUDE_FROM_ALL vcd_integrity_test.cpp)
target_link_libraries(edu_u2_vcd_integrity_test PRIVATE eda-wave-vcd)
add_executable(edu_u2_receipt_store_test EXCLUDE_FROM_ALL receipt_store_test.cpp)
target_link_libraries(edu_u2_receipt_store_test PRIVATE eda_agent_gateway Threads::Threads)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    get_filename_component(_u2_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(_dll libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
        if(EXISTS "${_u2_compiler_bin}/${_dll}")
            add_custom_command(TARGET edu_u2_gateway POST_BUILD COMMAND ${CMAKE_COMMAND} -E
                copy_if_different "${_u2_compiler_bin}/${_dll}" "$<TARGET_FILE_DIR:edu_u2_gateway>")
        endif()
    endforeach()
endif()
